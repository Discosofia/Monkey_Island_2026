function StateManager.GetState(character: Model): string
	return character:GetAttribute("State") or StateManager.States.Idle
end

-- Intentar cambiar de estado
function StateManager.SetState(character: Model, newState: string, duration: number?): boolean
	local currentState = StateManager.GetState(character)
	local currentPriority = Priority[currentState] or 0
	local newPriority = Priority[newState] or 0

	-- Comprobación de prioridad: Solo se acepta si la prioridad es igual o mayor
	if newPriority < currentPriority then
		return false
	end

	-- Limpiar timer anterior si existía
	if activeTimers[character] then
		task.cancel(activeTimers[character])
		activeTimers[character] = nil
	end

	-- Si el nuevo estado es de mayor prioridad (ej. Stun sobre Ataque), cancelamos animaciones activas
	if newPriority > currentPriority and currentPriority == Priority.Attacking then
		StateManager.CancelCurrentActions(character)
	end

	-- Aplicar nuevo atributo de estado
	character:SetAttribute("State", newState)

	-- Si tiene duración definida (ej. Stun de 1.5s o Parry de 0.3s), programa el retorno a Idle
	if duration and duration > 0 then
		activeTimers[character] = task.delay(duration, function()
			activeTimers[character] = nil
			-- Retornar a Idle una vez cumplido el tiempo
			character:SetAttribute("State", StateManager.States.Idle)
		end)
	end

	return true
end

-- Limpieza y detención de animaciones en curso al ser interrumpido
function StateManager.CancelCurrentActions(character: Model)
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if humanoid and humanoid:FindFirstChildOfClass("Animator") then
		local animator = humanoid:FindFirstChildOfClass("Animator")
		for _, track in animator:GetPlayingAnimationTracks() do
			if track.Priority == Enum.AnimationPriority.Action then
				track:Stop(0.1) -- Detención fluida de la animación de ataque [3]
			end
		end
	end
end

-- Tabla para rastrear tickets y velocidades originales por modelo
local activeStuns = {}

function StunHandler.ApplyStun(character: Model, duration: number)
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then return end

	-- Inicializar o recuperar registro del personaje
	if not activeStuns[character] then
		activeStuns[character] = {
			Ticket = 0,
			OriginalWalkSpeed = humanoid.WalkSpeed,
			OriginalJumpPower = humanoid.JumpPower
		}
	end

	local record = activeStuns[character]
	record.Ticket += 1
	local currentTicket = record.Ticket

	-- Cambiar estado a Stunned a través del StateManager
	StateManager.SetState(character, StateManager.States.Stunned)

	-- Restringir movimiento físico
	humanoid.WalkSpeed = 0
	humanoid.JumpPower = 0

	-- Programar la liberación de Stun
	task.delay(duration, function()
		-- Solo liberamos si no se ha recibido otro golpe que incrementó el ticket
		if activeStuns[character] and activeStuns[character].Ticket == currentTicket then
			-- Restaurar física original
			humanoid.WalkSpeed = record.OriginalWalkSpeed
			humanoid.JumpPower = record.OriginalJumpPower
			
			-- Retornar a estado Idle
			StateManager.SetState(character, StateManager.States.Idle)
			activeStuns[character] = nil
		end
	end)
end

return StunHandler

	end

	-- CASO 3: Impacto Limpio (Sin defensa)
	defenderHumanoid:TakeDamage(rawDamage) -- Respetar ForceFields
	StunHandler.ApplyStun(defender, stunDuration)
	
	VFXRemote:FireAllClients("HitEffect", defender.PrimaryPart.CFrame)

El método nativo de Roblox Studio para la detección por Raycast utiliza la función workspace:Raycast()12. local Players = game:GetService("Players")

local function RealizarAtaqueRaycast(attackerCharacter: Model, origen: Vector3, direccionObjetivo: Vector3, rangoMaximo: number)
	-- 1. Calcular el vector de dirección con la longitud/rango deseado
	-- La dirección debe ser un vector normalizado (.Unit) multiplicado por la distancia
	local direccionVector = direccionObjetivo.Unit * rangoMaximo

	-- 2. Configurar los parámetros del Raycast (RaycastParams)
	local raycastParams = RaycastParams.new()
	
	-- Filtrar al propio atacante para evitar que el rayo choque con su propio personaje
	raycastParams.FilterDescendantsInstances = { attackerCharacter }
	raycastParams.FilterType = Enum.RaycastFilterType.Exclude

	-- 3. Ejecutar la consulta en el Workspace
	local resultado: RaycastResult? = Workspace:Raycast(origen, direccionVector, raycastParams)

	-- 4. Evaluar el resultado de la colisión
	if resultado then
		local parteImpactada = resultado.Instance
		local posicionImpacto = resultado.Position
		local normalSuperficie = resultado.Normal

		-- Buscar si el objeto impactado pertenece al modelo de un personaje con Humanoid
		local characterModelo = parteImpactada:FindFirstAncestorOfClass("Model")
		if characterModelo then
			local humanoid = characterModelo:FindFirstChildWhichIsA("Humanoid")
			if humanoid and humanoid.Health > 0 then
				-- ¡Impacto confirmado contra un personaje!
				print("Objetivo impactado:", characterModelo.Name, "en posición:", posicionImpacto)
				return {
					Exito = true,
					Humanoid = humanoid,
					Posicion = posicionImpacto,
					Normal = normalSuperficie
				}
			end
		end

		print("El rayo impactó contra una superficie física/escenario:", parteImpactada.Name)
		return { Exito = false, Posicion = posicionImpacto, Normal = normalSuperficie }
	else
		-- Si no hubo colisión dentro del rango máximo, el punto final es la distancia máxima
		local puntoFinal = origen + direccionVector
		print("El ataque no impactó nada dentro del rango de", rangoMaximo, "studs.")
		return { Exito = false, Posicion = puntoFinal }
	end
end

La detección mediante Spherecast (una variante de Shapecasting en Roblox Studio) expande el raycasting tradicional de una línea unidimensional a una esfera tridimensional que barre el espacio a lo largo de un vector de dirección12.Es la técnica ideal para proyectiles con volumen (orbes de energía, bolas de fuego, flechas pesadas) o ataques cuerpo a cuerpo con área de impacto amplia, ya que evita que un proyectil veloz pase de largo entre fotogramas sin registrar impacto23.1. local SpherecastHitbox = {}

-- Realiza una consulta individual por Spherecast
function SpherecastHitbox.CastSphere(
	attackerCharacter: Model,
	origin: Vector3,
	radius: number,
	directionVector: Vector3
): { Exito: boolean, Humanoid: Humanoid?, Position: Vector3, Normal: Vector3, Instance: BasePart? }

	-- 1. Configurar los parámetros de filtro (RaycastParams)
	local params = RaycastParams.new()
	params.FilterDescendantsInstances = { attackerCharacter } -- Evita que el atacante se golpee a sí mismo
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.IgnoreWater = true

	-- 2. Ejecutar la consulta de Spherecast
	local result: RaycastResult? = Workspace:Spherecast(origin, radius, directionVector, params)

	-- 3. Procesar el resultado
	if result then
		local hitInstance = result.Instance
		local hitPosition = result.Position
		local hitNormal = result.Normal

		-- Buscar si el objeto tocado pertenece a un modelo con Humanoid
		local characterModel = hitInstance:FindFirstAncestorOfClass("Model")
		if characterModel then
			local humanoid = characterModel:FindFirstChildWhichIsA("Humanoid")
			if humanoid and humanoid.Health > 0 then
				return {
					Exito = true,
					Humanoid = humanoid,
					Character = characterModel,
					Position = hitPosition,
					Normal = hitNormal,
					Instance = hitInstance
				}
			end
		end

		-- Si impactó un obstáculo del mapa (pared, cobertura)
		return {
			Exito = false,
			Position = hitPosition,
			Normal = hitNormal,
			Instance = hitInstance
		}
	end

	-- Si no hubo colisión dentro de la distancia máxima
	return {
		Exito = false,
		Position = origin + directionVector,
		Normal = Vector3.yAxis,
		Instance = nil
	}
end

local StateManager = require(game:GetService("ReplicatedStorage").Modules.StateManager)
local StunHandler = require(script.Parent.StunHandler)
local VFXRemote = game:GetService("ReplicatedStorage"):WaitForChild("Events"):WaitForChild("CombatVFX")

function CombatResolver.ProcessHit(
	attacker: Model, 
	defender: Model, 
	rawDamage: number, 
	stunDuration: number, 
	isGuardBreak: boolean?
)
	local defenderState = StateManager.GetState(defender)
	local defenderHumanoid = defender:FindFirstChildOfClass("Humanoid")
	
	if not defenderHumanoid or defenderHumanoid.Health <= 0 then return end

	local isDefending = (defenderState == StateManager.States.Parrying or defenderState == StateManager.States.Blocking)

	-- CASO 1: El ataque es IMBLOQUEABLE (Guard Break)
	if isDefending and isGuardBreak then
		-- Romper la postura defensiva inmediatamente
		StateManager.SetState(defender, StateManager.States.Idle)
		
		-- Aplicar aturdimiento por rotura de guardia y daño completo
		StunHandler.ApplyStun(defender, stunDuration + 0.5) -- Penalización extra de tiempo
		defenderHumanoid:TakeDamage(rawDamage)
		
		VFXRemote:FireAllClients("GuardBreakEffect", defender.PrimaryPart.CFrame)
		return
	end

	-- CASO 2: Perfect Block / Parry Exitoso (Solo si NO es imbloqueable)
	if defenderState == StateManager.States.Parrying then
		StunHandler.ApplyStun(attacker, 1.2) -- Castigo al atacante
		VFXRemote:FireAllClients("ParryEffect", defender.PrimaryPart.CFrame)
		return
	end

	-- CASO 3: Bloqueo Normal (Solo si NO es imbloqueable)
	if defenderState == StateManager.States.Blocking then
		local reducedDamage = rawDamage * 0.2 -- 80% de reducción de daño
		defenderHumanoid:TakeDamage(reducedDamage)
		VFXRemote:FireAllClients("BlockEffect", defender.PrimaryPart.CFrame)
		return
	end

	-- CASO 4: Impacto Limpio (Incapaz de defenderse o sin bloqueo activo)
	defenderHumanoid:TakeDamage(rawDamage)
	StunHandler.ApplyStun(defender, stunDuration)
	VFXRemote:FireAllClients("HitEffect", defender.PrimaryPart.CFrame)
end

return CombatResolver

local Workspace = game:GetService("Workspace")
local CombatResolver = require(script.Parent.CombatResolver)

local ProjectileService = {}

type ProjectileData = {
	Attacker: Model,
	Origin: Vector3,
	Direction: Vector3,
	Speed: number,
	Radius: number,
	MaxDistance: number,
	Damage: number,
	StunDuration: number,
	IsGuardBreak: boolean
}

function ProjectileService.Fire(data: ProjectileData)
	local currentPos = data.Origin
	local distanceTraveled = 0
	local connection: RBXScriptConnection? = nil

	-- Configurar filtro de colisión (Ignorar al atacante)
	local raycastParams = RaycastParams.new()
	raycastParams.FilterDescendantsInstances = { data.Attacker }
	raycastParams.FilterType = Enum.RaycastFilterType.Exclude
	raycastParams.IgnoreWater = true

	connection = RunService.Heartbeat:Connect(function(deltaTime)
		-- 1. Calcular el vector de desplazamiento para este fotograma
		local stepDistance = data.Speed * deltaTime
		local directionVector = data.Direction.Unit * stepDistance

		-- 2. Ejecutar Spherecast a lo largo del paso del fotograma
		local hitResult: RaycastResult? = Workspace:Spherecast(currentPos, data.Radius, directionVector, raycastParams)

		if hitResult then
			local hitInstance = hitResult.Instance
			local hitModel = hitInstance:FindFirstAncestorOfClass("Model")

			-- Evaluar si se impactó a un personaje con Humanoid
			if hitModel then
				local humanoid = hitModel:FindFirstChildWhichIsA("Humanoid")
				if humanoid and humanoid.Health > 0 and hitModel ~= data.Attacker then
					-- Procesar impacto con la regla de Guard Break
					CombatResolver.ProcessHit(
						data.Attacker,
						hitModel,
						data.Damage,
						data.StunDuration,
						data.IsGuardBreak
					)
					connection:Disconnect()
					return
				end
			end

			-- Impacto contra una pared o estructura del mapa
			print("El proyectil chocó contra la cobertura:", hitInstance.Name)
			connection:Disconnect()
			return
		end

		-- 3. Actualizar posición y verificar límite de alcance
		currentPos += directionVector
		distanceTraveled += stepDistance

		if distanceTraveled >= data.MaxDistance then
			connection:Disconnect()
		end
	end)
end

return ProjectileService

local CooldownManager = {}

-- Estructura de almacenamiento: [UserId][AbilityName] = { LastUsed = number, Duration = number }
local activeCooldowns = {}

-- Tabla de tiempos base por habilidad
local BaseCooldowns = {
	M1 = 0.5,
	LightAttack = 2.0,
	HeavyAttack = 6.0,
	Fireball = 8.0,
	Dash = 4.0,
	Parry = 5.0
}

-- Inicializar registro para un jugador
function CooldownManager.InitPlayer(player: Player)
	activeCooldowns[player.UserId] = {}
end

-- Limpieza al salir el jugador
function CooldownManager.RemovePlayer(player: Player)
	activeCooldowns[player.UserId] = nil
end

-- Verifica si una habilidad se puede usar (Retorna boolean y tiempo restante)
function CooldownManager.CanUse(player: Player, abilityName: string): (boolean, number)
	local playerTable = activeCooldowns[player.UserId]
	if not playerTable then return true, 0 end

	local data = playerTable[abilityName]
	if not data then return true, 0 end

	local timePassed = os.clock() - data.LastUsed
	local timeRemaining = data.Duration - timePassed

	if timeRemaining <= 0 then
		return true, 0
	else
		return false, math.max(0, timeRemaining)
	end
end

-- Establecer o reiniciar el Cooldown
function CooldownManager.SetCooldown(player: Player, abilityName: string, customDuration: number?, modifierPercent: number?)
	local userId = player.UserId
	if not activeCooldowns[userId] then
		activeCooldowns[userId] = {}
	end

	-- Obtener duración base o personalizada
	local duration = customDuration or BaseCooldowns[abilityName] or 1.0

	-- Aplicar modificador opcional (ejemplo: 0.8 para un 20% de reducción)
	if modifierPercent then
		duration = duration * modifierPercent
	end

	-- Guardar marca de tiempo
	activeCooldowns[userId][abilityName] = {
		LastUsed = os.clock(),
		Duration = duration
	}

	-- Replicar al cliente para animar la UI (HUD / Hotbar)
	CooldownRemote:FireClient(player, abilityName, duration)
end

-- Reducir el tiempo restante de forma plana (ej. por acertar un combo o Parry)
function CooldownManager.ReduceCooldown(player: Player, abilityName: string, secondsToReduce: number)
	local userId = player.UserId
	if activeCooldowns[userId] and activeCooldowns[userId][abilityName] then
		local data = activeCooldowns[userId][abilityName]
		data.LastUsed -= secondsToReduce -- Restar tiempo a la marca inicial acelera el fin
		
		-- Recalcular y enviar actualización a la UI
		local newRemaining = math.max(0, data.Duration - (os.clock() - data.LastUsed))
		CooldownRemote:FireClient(player, abilityName, newRemaining)
	end
end

return CooldownManager

local function IsHitValid(attackerPlayer: Player, characterToDamage: Model, hitPosition: Vector3, originPos: Vector3): boolean
	-- 1. Comprobar la distancia entre la posición reportada y la ubicación actual del objetivo
	local hrp = characterToDamage:FindFirstChild("HumanoidRootPart") :: BasePart
	if not hrp then return false end

	local distance = (hrp.Position - hitPosition).Magnitude
	if distance > MAX_HIT_PROXIMITY then
		return false -- El objetivo ya no estaba en esa posición
	end

	-- 2. Comprobar Línea de Visión (Line of Sight) para evitar disparar a través de muros
	local rayLength = (hitPosition - originPos).Magnitude
	local rayDirection = (hitPosition - originPos).Unit
	
	local rayParams = RaycastParams.new()
	rayParams.FilterDescendantsInstances = { attackerPlayer.Character }
	rayParams.FilterType = Enum.RaycastFilterType.Exclude

	local rayResult = Workspace:Raycast(originPos, rayDirection * rayLength, rayParams)
	if rayResult and not rayResult.Instance:IsDescendantOf(characterToDamage) then
		return false -- Un obstáculo del mapa bloqueó el disparo
	end

	return true
end

local function SpawnServerProjectile(attackerPlayer: Player, origin: Vector3, directionVector: Vector3, speed: number, radius: number)
	local currentPos = origin
	local distanceTraveled = 0
	local maxDistance = directionVector.Magnitude
	local directionUnit = directionVector.Unit

	local rayParams = RaycastParams.new()
	rayParams.FilterDescendantsInstances = { attackerPlayer.Character }
	rayParams.FilterType = Enum.RaycastFilterType.Exclude

	-- Notificar a los DEMÁS clientes para que muestren la bala visualmente sin retardo local
	replicateVFXRemote:FireAllClients(attackerPlayer, origin, directionVector, speed)

	local connection: RBXScriptConnection? = nil
	connection = RunService.Heartbeat:Connect(function(deltaTime)
		local stepDistance = speed * deltaTime
		local stepVector = directionUnit * stepDistance

		-- Barredura volumétrica entre el fotograma anterior y el actual
		local hitResult: RaycastResult?
		if radius > 0 then
			hitResult = Workspace:Spherecast(currentPos, radius, stepVector, rayParams)
		else
			hitResult = Workspace:Raycast(currentPos, stepVector, rayParams)
		end

		if hitResult then
			local hitInstance = hitResult.Instance
			local hitPosition = hitResult.Position
			local characterModel = hitInstance:FindFirstAncestorOfClass("Model")

			if characterModel then
				local humanoid = characterModel:FindFirstChildWhichIsA("Humanoid")
				if humanoid and humanoid.Health > 0 and characterModel ~= attackerPlayer.Character then
					-- Validar antes de infligir daño
					if IsHitValid(attackerPlayer, characterModel, hitPosition, origin) then
						humanoid:TakeDamage(25) -- TakeDamage respeta ForceFields de spawn
					end
				end
			end

			connection:Disconnect()
			return
		end

		currentPos += stepVector
		distanceTraveled += stepDistance

		if distanceTraveled >= maxDistance then
			connection:Disconnect()
		end
	end)
end

local CombatResolver = {}

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local StateManager = require(ReplicatedStorage.Modules.StateManager)
local StunHandler = require(script.Parent.StunHandler)

local VFXRemote = ReplicatedStorage:WaitForChild("Events"):WaitForChild("CombatVFX")

function CombatResolver.ProcessHit(
	attacker: Model,
	defender: Model,
	rawDamage: number,
	stunDuration: number,
	isUnblockable: boolean? -- Flag que indica si el ataque es imbloqueable
)
	local defenderState = StateManager.GetState(defender)
	local defenderHumanoid = defender:FindFirstChildOfClass("Humanoid")

	if not defenderHumanoid or defenderHumanoid.Health <= 0 then return end

	local isDefending = (defenderState == StateManager.States.Parrying or defenderState == StateManager.States.Blocking)

	---------------------------------------------------------
	-- CASO 1: ATAQUE IMBLOQUEABLE (GUARD BREAK)
	---------------------------------------------------------
	if isDefending and isUnblockable then
		-- 1. Cancelar inmediatamente la postura defensiva
		StateManager.SetState(defender, StateManager.States.Idle)

		-- 2. Aplicar aturdimiento extendido por rotura de guardia
		local guardBreakStunTime = stunDuration + 0.6
		StunHandler.ApplyStun(defender, guardBreakStunTime)

		-- 3. Infligir daño completo (respetando ForceFields)
		defenderHumanoid:TakeDamage(rawDamage)

		-- 4. Notificar a los clientes para efectos de rotura de guardia
		VFXRemote:FireAllClients("GuardBreakEffect", defender.PrimaryPart.CFrame)
		return
	end

	---------------------------------------------------------
	-- CASO 2: PERFECT BLOCK / PARRY EXITOSO (Ataques bloqueables)
	---------------------------------------------------------
	if defenderState == StateManager.States.Parrying then
		StunHandler.ApplyStun(attacker, 1.2) -- Castigo al atacante
		VFXRemote:FireAllClients("ParryEffect", defender.PrimaryPart.CFrame)
		return
	end

	---------------------------------------------------------
	-- CASO 3: BLOQUEO NORMAL (Ataques bloqueables)
	---------------------------------------------------------
	if defenderState == StateManager.States.Blocking then
		local reducedDamage = rawDamage * 0.2 -- 80% de reducción de daño
		defenderHumanoid:TakeDamage(reducedDamage)
		VFXRemote:FireAllClients("BlockEffect", defender.PrimaryPart.CFrame)
		return
	end

	---------------------------------------------------------
	-- CASO 4: IMPACTO LIMPIO (Objetivo sin bloqueo activo)
	---------------------------------------------------------
	defenderHumanoid:TakeDamage(rawDamage)
	StunHandler.ApplyStun(defender, stunDuration)
	VFXRemote:FireAllClients("HitEffect", defender.PrimaryPart.CFrame)
end

return CombatResolver

### 1\. Estructura Esencial de un Combo

Un combo fluido y bien diseñado suele estructurarse en **tres fases principales** intercaladas con ataques básicos (M1)[5][6]:

* **Apertura o Iniciador (** **Opener** **):** Habilidades rápidas de desplazamiento, agarre o aturdimiento ligero que inician el combo[5]. Si el atacante falla la apertura, debe quedar expuesto a un tiempo de recuperación (*endlag*) para que el oponente pueda castigarlo[7][8].
* **Extensor (** **Extender** **):** Movimientos que extienden el estado de aturdimiento (*stun*) o reubican al rival cerca del jugador (mediante *M1s*, empujones o enlaces de habilidades)[5].
* **Rematador (** **Finisher** **):** Ataques de alto impacto o derribo final que infligen el mayor daño cuando el oponente ya no puede esquivar[5][7].

---

### 2\. Mecánicas de Combate y Tácticas Avanzadas (*Techs*)

Para dar profundidad táctica y permitir que los jugadores desarrollen habilidad propia, es crucial implementar mecánicas de interacción directa[6]:

* **Intercambio de Ataques Básicos (** **M1 Trading** **):** Los golpes básicos (*M1*) deben aplicar un breve bamboleo (*stagger*)[6]. El jugador que gestione mejor los tiempos de recuperado o los saltos al atacar gana la iniciativa para iniciar su combo[6].
* **Cancelaciones de Animación (** **Roll/Spell Canceling** **):**
  * Permitir cancelar esquives (*roll cancel*) consumiendo recursos para ganar cuadros de invulnerabilidad (*i-frames*) y posicionarse rápidamente[9].
  * Implementar la cancelación de habilidades o hechizos (*spell/mantra canceling*) para amagar (*faint*), engañar la reacción del rival y hacerle gastar sus recursos defensivos[9].
* **Impacto contra el Entorno (** **Wall Bangs / Wall Combos** **):** Aprovechar la geometría del mapa para que los ataques que empujan al rival reboten contra paredes o esquinas, permitiendo encadenar extensiones adicionales[12].
* **Gestión de Esquive y Bloqueo (** **Instinct Break &amp; Evasives** **):**
  * Sistemas de **esquive limitado o Visión (** **Instinct** **)** que evitan ataques automáticos pero se rompen tras recibir cierto número de *M1s*, ataques de área (AoE) o movimientos específicos de ruptura (*Instinct break*)[16][17].
  * Barra de **Escape/Evasivo (** **Ragdoll Cancel** **)**: Recursos defensivos con tiempo de recarga que permiten al jugador aturdido escapar de un combo para evitar ser eliminado de un solo turno (*100-to-0 true combo*) sin capacidad de respuesta[18].
* **Bloqueo de Pánico (** **Scared Parry** **):** Si un jugador intenta hacer *parry* o bloqueo mientras ya está en aturdimiento por fallar una lectura, entra en una animación de pánico que lo deja indefenso por un breve lapso[22].

---

### 3\. Estilos de Juego Principal (*Playstyles*)

Diseñar clases, frutas, armas o artes marciales debe responder a diferentes arquetipos de jugador[2]:

1. **Agresivo (** **Rushdown / Presión Directa** **):** Personajes o equipamientos orientados al combate cuerpo a cuerpo veloz, reducción de distancias y combos continuos de alto daño[4].
2. **Defensivo / Contraataque (** **Counter-Play / Spacing** **):** Basado en mantener la distancia, esperar el fallo o *endlag* del rival, ejecutar bloqueos/parries precisos (*perfect dodge*) y castigar inmediatamente[2].
3. **Control de Espacio / Distancia (** **Zoning &amp; Range** **):** Diseñado para manipular la posición del oponente con proyectiles, trampas o invocaciones desde ángulos múltiples antes de entrar a cerrar la baja[8].
4. **Técnico / Fundamentos Básicos (** **Boxing / Grounded Fighting** **):** Enfocado en la gestión de estamina, amagos (*feints*), posicionamiento y lectura táctica del rival sin depender de habilidades mágicas masivas[2].

---

### 4\. Mejores Prácticas de Diseño e Implementación

* **Evitar los** **True Combos** **Ineludibles Infiinitos:** Garantiza siempre una ventana defensiva (desactivación de *Instinct*, uso de *evasive* o resistencia de *posture*) para que el combate sea una conversación de ida y vuelta y no un castigo unidireccional[16].
* **Riesgo vs. Recompensa en los Iniciadores:** Un *opener* con gran alcance o aturdimiento largo debe conllevar un tiempo de reutilización (*cooldown*) extenso o dejar al usuario muy vulnerable si falla[7].
* **Legibilidad Visual y Sonora:** Añadir señales claras (efectos de sonido o animaciones distintivas) cuando alguien amaga o cancela un ataque, permitiendo que los jugadores con mejores reflejos puedan reaccionar[8][9].

El equilibrio del daño en el combate PvP se logra mediante un conjunto de **multiplicadores específicos**, **mecanismos de prevención de aturdimiento infinito (** **anti-stunlock** **)** y un **diseño horizontal de equipamiento**:

### 1\. Multiplicadores Directos de PvP y Target TTK

En lugar de utilizar los mismos valores de daño que contra los enemigos del entorno (PvE), los enfrentamientos PvP aplican ajustes a la escala de combate[1][2]:

* **Daño de Armas:** Se reduce a un **×0,55** (con excepciones como el arco a ×0,7)[2].
* **Habilidades Activas:** Se ajustan a un **×0,45** de su valor base[2].
* **Curaciones y Escudos:** Se escalan a un **×0,6**[2].
* **Tiempo para Eliminar (TTK):** Estas reducciones están calibradas para que un duelo entre dos roles equivalentes (como Guerrero, Cazador o Mago) dure en promedio entre **8 y 10 segundos**[2]. Esto evita eliminaciones instantáneas de un solo turno (*100-to-0 true combos*) y da espacio a la recuperación táctica.

---

### 2\. Control de Masas e Inmunidad a Aturdimientos (*Stunlock*)

Para evitar que un jugador quede indefenso si es atacado por varios rivales o por secuencias continuas:

* **Duración de Control al 50%:** Todos los aturdimientos, congelaciones, agarres e inmovilizaciones duran la mitad de tiempo (**×0,5**) en PvP respecto a su versión PvE[1][2].
* **Ventana de Inmunidad (** **PlayerControlImmunitySeconds** **):** Tras ser liberado de cualquier efecto de control o agarre, el personaje recibe **2 segundos de inmunidad a nuevos aturdimientos**[3]. Durante este lapso puede moverse y responder, aunque sigue recibiendo daño normal de ataques[3].

---

### 3\. Ajuste de Ritmo de Habilidades (Frecuencia vs. Daño)

Para lograr un combate ágil donde los combos fluyan sin inflar el daño por segundo (DPS) total:

* Las habilidades activas reducen sus tiempos de recarga un **40%**, pero su daño e impacto individual se reducen en ese mismo **40%**[4].
* Esto permite enlazar habilidades con mayor frecuencia para formar combos sin que el daño de una sola habilidad resulte devastador[4].

---

### 4\. Equipamiento Horizontal y Techos de Estadísticas (*Soft Caps*)

* **Variantes Horizontales:** El equipamiento especial (comprado con Ámbar en la arena) no ofrece ventajas numéricas absolutas, sino intercambios tácticos (*trade-offs*)[5]. Por ejemplo, aumentar el área de impacto o el daño a cambio de un tiempo de recarga más largo[5].
* **Límites de Rendimiento (** **Soft Caps** **):** Al igual que en otros títulos competitivos de la plataforma, el sistema equilibra los puntos de vida y daño con rendimientos decrecientes a partir de ciertos umbrales, impidiendo la creación de personajes invencibles o destructores de un solo golpe

### 1\. Sistema Neutral Defensivo: *Parry* y Amagos (*Feints*)

* **Estado Actual en el GDD:** El juego cuenta con esquiva/dash, multiplicadores reducidos en PvP (×0,45 en habilidades, ×0,55 en armas)[5] y una ventana de inmunidad al control tras ser liberado (`PlayerControlImmunitySeconds`)[5].
* **Lo que enseñan los referentes:** En títulos como *Deepwoken* y *The Strongest Battlegrounds*, la base del combate de alta habilidad radica en la lectura del rival mediante **bloqueos perfectos (** **Parry** **)** y **amagos (** **Feints / Spell Canceling** **)**[6]. Los amagos permiten cancelar la animación de un ataque o hechizo consumiendo recurso para engañar la reacción del oponente y hacerle gastar sus opciones defensivas[7][8].
* **Oportunidad de Mejora:**
  * Implementar una ventana breve de **Bloqueo Perfecto (** **Parry** **)**: si el jugador bloquea justo cuando entra el golpe, no recibe daño y desequilibra al atacante por un instante[6][9].
  * Permitir **cancelar la animación de inicio de habilidades** o ataques cargados (presionando esquiva o bloqueo durante el inicio)[7][10]. Esto evita que el combate se reduzca a lanzar habilidades en recarga y premia el engaño táctico.

### 2\. Mecánicas Anti-Bloqueo (*Guard Break / Instinct Break*)

* **Estado Actual en el GDD:** Los aturdimientos y controles se reducen al 50% en PvP[5], pero no se especifica cómo responder contra un jugador que mantiene el bloqueo constante.
* **Lo que enseñan los referentes:** En *Blox Fruits*, la mecánica de *Instinct Break* obliga a usar movimientos específicos de ruptura o cadenas de ataques básicos para consumir los esquives automáticos del rival antes de iniciar el combo[11][12]. En *The Strongest Battlegrounds*, ciertos ataques pesados rompen el bloqueo directamente (*Guard Break*)[9][13].
* **Oportunidad de Mejora:**
  * Designar ciertas habilidades activas (como el *Rompehuesos* del Guerrero o la *Flecha perforante* del Cazador)[14] con la propiedad de **Ruptura de Bloqueo**.
  * Si un rival mantiene el bloqueo activo contra una habilidad de este tipo, su postura se rompe (*stagger* pesado), dejándolo vulnerable a la primera habilidad iniciadora (*Opener*) de un combo[9][15].

### 3\. Recurso de Escape Activo (*Ragdoll Cancel / Evasivo*)

* **Estado Actual en el GDD:** El sistema aplica inmunidad automática de 2 segundos tras salir de un aturdimiento[5] e intercala las habilidades en un bucle A → B → A mediante conexiones[15].
* **Lo que enseñan los referentes:** En *Jujutsu Shenanigans* y *Blox Fruits*, la libertad de escape no es solo pasiva, sino un **recurso activo (** **Evasive / Ragdoll Cancel / V4** **)**[3]. Permite interrumplir un combo devastador a cambio de un tiempo de recarga largo (ej. 30–40 s) o consumo de estamina.
* **Oportunidad de Mejora:**
  * Otorgar a cada rol un **Escape de Pánico (Evasivo)** con un tiempo de recarga independiente alto. Esto permite liberarse en medio de una cadena de aturdimiento antes de perder toda la barra de vida, obligando al atacante a baitear (*engañar*) el evasivo rival antes de lanzar su combo completo[3].

### 4\. Interacción con el Entorno (*Wall Bangs / Rebotes*)

* **Lo que enseñan los referentes:** Tanto *The Strongest Battlegrounds* como *Deepwoken* utilizan la geometría del mapa a través de los **"Wall Bangs"**: si un impacto que empuja proyecta al enemigo contra una pared o estructura, este rebota y queda aturdido por una fracción de segundo adicional, permitiendo extender el combo[20][21].
* **Oportunidad de Mejora:**

Esto convierte la posición espacial en el mapa en un factor estratégico fundamental.

### 1\. Sistema de Parry (Bloqueo Perfecto)

El *Parry* premia la reacción precisa sobre el bloqueo pasivo continuo.

* **Ventana de Reacción (** **Parry Window** **):** Al activar la tecla de defensa (**F** en PC / **L2** en Mando / **Botón de Escudo** en Móvil), el personaje entra en una postura de guardia activa durante **0.25 segundos (15 frames)**.
* **Recompensa al Acertar (** **Perfect Parry** **):**
  * **Inmunidad total:** Cancela el 100% del daño del ataque entrante (tanto golpes M1 como habilidades esquivables)[2].
  * **Desequilibrio al Atacante (** **Stagger** **):** Corta la animación del rival y le aplica un aturdimiento de **0.35 s en PvP** (0.8 s contra monos/jefes en PvE)[2].
  * **Sonido y Visual:** Un destello dorado metálico con sonido seco de choque (*clash*).
* **Bloqueo Normal vs. Parry:**
  * Si se mantiene presionada la tecla después de los 0.25 s iniciales, la postura pasa a **Bloqueo Normal**.
  * El Bloqueo Normal reduce el daño recibido un 50%, pero consume barra de **Postura/Estamina** por cada impacto recibido[4]. Si la postura llega a cero, la guardia se rompe (*Guard Break*).
* **Castigo por Fallo (** **Scared Parry / Bloqueo de Pánico** **):**
  * Si el jugador intenta hacer *parry* a destiempo o presiona el botón mientras ya está aturdido por un combo, entra en una animación de **pánico de 0.5 s**[5].
  * Durante este estado no se puede esquivar ni bloquear, quedando totalmente expuesto al contraataque rival[5].

### 2\. Sistema de Amagos (*Feints / Cancels*)

Los amagos permiten engañar la lectura del oponente para forzar un *Parry* a destiempo o hacerle gastar sus habilidades defensivas[3][6].

* **Amago de Habilidad (** **Mantra / Spell Canceling** **):**
  * **Mecánica:** Presionar **Bloqueo (F / L2)** durante la animación de inicio (*windup*, primeros 0.20 s) de cualquier habilidad activa (Z, X, C, R)[3][6].
  * **Efecto:** El personaje interrumpe el lanzamiento produciendo el sonido e inicio visual del ataque, engañando al rival[2][3].
  * **Costo:** Consume un **15% de Estamina** y aplica un tiempo de recarga corto de penalización (**1.5 segundos**) a la habilidad amagada en lugar de su *cooldown* completo[3].
* **Amago de Golpe Básico (** **M1 Feint** **):**
  * Interrumpe el inicio del swing de un ataque M1 presionando Bloqueo[7]. Sirve para baitear la esquiva del rival y castigarlo en el punto de aterrizaje[8].
* **Cancelación de Esquive (** **Roll Cancel** **):**
  * Permite cortar la animación del rodaje/dash presionando el botón de ataque M1 o habilidad en cualquier punto del recorrido[2].
  * Otorga cuadros de invulnerabilidad (*i-frames*) durante la esquiva y permite reaparecer atacando inmediatamente desde un ángulo ciego del oponente[2].

### 1\. Pilares del Sistema de Interacción con el Entorno

1. **Rebote contra Muros y Estructuras (** **Wall Bangs / Wall Combos** **):**

2. **Coberturas y Obstáculos Destruibles (** **Destructible Objects** **):**
  * **En la Arena de Dunas:** Columnas de piedra erosionadas, vallas de madera y altares pequeños pueden agrietarse y colapsar tras recibir habilidades de alto impacto (como *Meteoro* o *Pisotón*), eliminando coberturas estratégicas del rival[1][4].
  * **En el Basecamp (PvE):** Las **estacas defensivas** dañan a los monos o jugadores rivales que sean empujados contra ellas[3].
3. **Terrenos Modificados por Habilidades (** **Hazard Zones** **):**
  * Áreas afectadas temporalmente por habilidades: el suelo ardiendo por la *Lluvia de estrellas* del Mago (daño continuo por 3 s)[5], la congelación de la *Nova de hielo*[5] o los *Cepos* del Cazador[6].

### 2\. Flujo Lógico de Detección de *Wall Bang*

Para evitar falsos positivos y asegurar que el rendimiento sea óptimo en móviles y PC, la colisión se calcula en el servidor mediante **Raycasting direccional** durante la trayectoria del empuje:

```
[Inicio de Knockback] ──&gt; [Desplazamiento continuo por LinearVelocity]
                                 │
                        (En cada Step de física)
                                 │
                   ¿Raycast detecta pared a &lt; 3 studs?
                         ├── SÍ  ──&gt; 1. Detener LinearVelocity
                         │           2. Activar animación de choque (Ragdoll breve)
                         │           3. Aplicar Wall Stun (0.35 s)
                         │           4. Emitir efecto visual de grieta/polvo
                         │
                         └── NO  ──&gt; Continuar empuje normal hasta frenar

```

### 3\. Implementación del Código en Luau (Roblox Studio)

Este script modular maneja el vector de empuje y la detección de impacto contra el entorno utilizando `RaycastParams`:

```
-- ServerScriptService/Combat/EnvironmentInteractionService.luau
local Workspace = game:GetService("Workspace")
local TweenService = game:GetService("TweenService")

local EnvironmentService = {}

-- Configuración de Raycast que ignora personajes y efectos visuales
local raycastParams = RaycastParams.new()
raycastParams.FilterType = Enum.RaycastFilterType.Exclude
raycastParams.IgnoreWater = true

function EnvironmentService.ApplyKnockbackWithWallCheck(targetCharacter, directionVector, force, duration)
    local hrp = targetCharacter:FindFirstChild("HumanoidRootPart")
    local humanoid = targetCharacter:FindFirstChild("Humanoid")
    if not hrp or not humanoid then return end

    -- Ignorar a todos los personajes en la detección de paredes
    local ignoreList = {}
    for _, player in ipairs(game.Players:GetPlayers()) do
        if player.Character then table.insert(ignoreList, player.Character) end
    end
    raycastParams.FilterObjects = ignoreList

    -- Crear impulso físico
    local linearVelocity = Instance.new("LinearVelocity")
    local attachment = Instance.new("Attachment", hrp)
    linearVelocity.Attachment0 = attachment
    linearVelocity.MaxForce = 100000
    linearVelocity.VectorVelocity = directionVector.Unit * force
    linearVelocity.Parent = hrp

    local startTime = tick()
    local wallHit = false

    -- Bucle de detección frame a frame durante el empuje
    local connection
    connection = game:GetService("RunService").Heartbeat:Connect(function()
        local elapsedTime = tick() - startTime
        
        -- Detección frontal de pared (Raycast 3 studs adelante en la dirección del empuje)
        local rayResult = Workspace:Raycast(hrp.Position, directionVector.Unit * 3, raycastParams)

        if rayResult and rayResult.Instance and rayResult.Instance.CanCollide then
            wallHit = true
            connection:Disconnect()
            linearVelocity:Destroy()
            attachment:Destroy()

            -- Ejecutar efecto de Wall Bang
            EnvironmentService.TriggerWallBang(targetCharacter, rayResult.Position, rayResult.Normal)
        elseif elapsedTime &gt;= duration then
            connection:Disconnect()
            linearVelocity:Destroy()
            attachment:Destroy()
        end
    end)
end

function EnvironmentService.TriggerWallBang(character, hitPosition, hitNormal)
    local humanoid = character:FindFirstChild("Humanoid")
    if not humanoid then return end

    -- 1. Aplicar estado de Wall Stun breve (0.35 segundos)
    character:SetAttribute("WallStunning", true)
    humanoid:ChangeState(Enum.HumanoidStateType.Physics) -- Animación de choque

    -- 2. Emitir efectos visuales/sonoros en el punto de impacto
    local impactFX = Instance.new("Part")
    impactFX.Anchored = true
    impactFX.CanCollide = false
    impactFX.Size = Vector3.new(1, 1, 1)
    impactFX.CFrame = CFrame.lookAt(hitPosition, hitPosition + hitNormal)
    impactFX.Parent = Workspace

    -- Limpieza y restauración tras el Wall Stun
    task.delay(0.35, function()
        if character and character:Parent() then
            character:SetAttribute("WallStunning", nil)
            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        impactFX:Destroy()
    end)
end

return EnvironmentService

```

### 4\. Equilibrio y Reglas de Diseño

* **Límite de Rebote (** **Wall Bang Cooldown** **):** Para evitar que un jugador quede atrapado en una esquina recibiendo rebotes infinitos, se aplica un temporizador de **1.5 segundos de inmunidad a** **Wall Bangs** tras sufrir el primer choque.
* **Posicionamiento Táctico en Dunas:** En la arena 5v5, atraer al portador de la bandera hacia los pasillos laterales o ruinas cerradas aumenta el riesgo de sufrir un *Wall Bang*, mientras que avanzar por el centro de la pirámide ofrece mayor espacio pero menor cobertura

### 1\. Pilares Mecánicos de Interacción

* **Rebote contra Muros (** **Wall Bangs / Wall Combos** **):** Cuando una habilidad con empuje (*knockback*), como la *Embestida* del Guerrero (20 studs)[5] o la repulsión de las habilidades del Cazador[5], proyecta a un rival contra una estructura sólida (paredes de roca, ruinas mayas en Dunas o empalizadas del Basecamp)[2]:
  1. El desplazamiento del objetivo se detiene al colisionar contra la superficie[4].
  2. Se aplica un breve aturdimiento de impacto (**0.35 s de** **Wall Stun**)[4].
  3. Se activan partículas de polvo/grietas y un efecto sonoro de impacto.
* **Objetos y Coberturas en Dunas:** En la arena 5v5 de la Isla de Dunas (mapa de 400 × 200 studs con cañones y ruinas)[2], los ataques de alto impacto en área (*Meteoro* del Mago, *Pisotón* del Guerrero)[5] pueden agrietar o romper coberturas menores (columnas rotas, estelas)[2], abriendo líneas de visión para los atacantes a distancia[3].
* **Alteración del Terreno (** **Hazard Zones** **):** Habilidades como la *Lluvia de estrellas* del Mago dejan el suelo ardiendo durante 3 segundos[5], la *Nova de hielo* congela superficies[5] y el *Cepo* del Cazador atrapa a quienes lo pisan[5], creando áreas de control de espacio (*zoning*)[6].
* **Peligros Ambientales en el Basecamp (PvE):** Las defensas y estacas del Basecamp dañan a los monos o jugadores rivales empujados contra ellas[1].

### 2\. Flujo de Colisión y Física en Servidor

Para garantizar un rendimiento fluido en dispositivos **Móvil y PC**[7], la colisión se calcula en el servidor mediante **Raycasting direccional** durante la trayectoria del empuje[8]:

```
[Inicio de Knockback] ──&gt; [Desplazamiento por LinearVelocity]
                                 │
                   (Raycast continuo en RunService.Heartbeat)
                                 │
                   ¿Detecta pared CanCollide a &lt; 3 studs?
                         ├── SÍ  ──&gt; 1. Cancelar LinearVelocity
                         │           2. Estado Physics / Ragdoll breve
                         │           3. Aplicar Wall Stun (0.35 s)
                         │           4. Emitir partículas e impacto
                         │
                         └── NO  ──&gt; Continuar trayectoria hasta frenar

```

### 3\. Código de Implementación en Luau (Roblox Studio)

A continuación se muestra el módulo ejecutable en el servidor para gestionar empujes y detección de estructuras:

```
-- ServerScriptService/Combat/EnvironmentService.luau
local Workspace = game:GetService("Workspace")
local RunService = game:GetService("RunService")

local EnvironmentService = {}

local raycastParams = RaycastParams.new()
raycastParams.FilterType = Enum.RaycastFilterType.Exclude
raycastParams.IgnoreWater = true

function EnvironmentService.ApplyKnockbackWithWallCheck(targetCharacter, directionVector, force, duration)
    local hrp = targetCharacter:FindFirstChild("HumanoidRootPart")
    local humanoid = targetCharacter:FindFirstChild("Humanoid")
    if not hrp or not humanoid then return end

    -- Excluir personajes del Raycast para detectar solo estructuras sólidas
    local ignoreList = {}
    for _, player in ipairs(game.Players:GetPlayers()) do
        if player.Character then table.insert(ignoreList, player.Character) end
    end
    raycastParams.FilterObjects = ignoreList

    -- Aplicar impulso de física
    local attachment = Instance.new("Attachment", hrp)
    local linearVelocity = Instance.new("LinearVelocity")
    linearVelocity.Attachment0 = attachment
    linearVelocity.MaxForce = 100000
    linearVelocity.VectorVelocity = directionVector.Unit * force
    linearVelocity.Parent = hrp

    local startTime = tick()
    local connection

    connection = RunService.Heartbeat:Connect(function()
        local elapsedTime = tick() - startTime
        
        -- Detección frontal a 3 studs en la dirección del empuje
        local rayResult = Workspace:Raycast(hrp.Position, directionVector.Unit * 3, raycastParams)

        if rayResult and rayResult.Instance and rayResult.Instance.CanCollide then
            connection:Disconnect()
            linearVelocity:Destroy()
            attachment:Destroy()

            -- Desencadenar efecto de Wall Bang
            EnvironmentService.TriggerWallBang(targetCharacter, rayResult.Position, rayResult.Normal)
        elseif elapsedTime &gt;= duration then
            connection:Disconnect()
            linearVelocity:Destroy()
            attachment:Destroy()
        end
    end)
end

function EnvironmentService.TriggerWallBang(character, hitPosition, hitNormal)
    local humanoid = character:FindFirstChild("Humanoid")
    if not humanoid then return end

    -- 1. Aplicar estado de Wall Stun (0.35 s)
    character:SetAttribute("WallStunning", true)
    humanoid:ChangeState(Enum.HumanoidStateType.Physics)

    -- 2. Emitir partículas y sonido en el punto de colisión
    local impactPoint = Instance.new("Part")
    impactPoint.Anchored = true
    impactPoint.CanCollide = false
    impactPoint.Size = Vector3.new(1, 1, 1)
    impactPoint.CFrame = CFrame.lookAt(hitPosition, hitPosition + hitNormal)
    impactPoint.Parent = Workspace

    -- Limpieza y recuperación
    task.delay(0.35, function()
        if character and character:Parent() then
            character:SetAttribute("WallStunning", nil)
            humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        impactPoint:Destroy()
    end)
end

return EnvironmentService

```

### 4\. Reglas de Balance e Inmunidad

* **Inmunidad a Bucles de Rebote:** Tras sufrir un *Wall Bang*, el objetivo recibe **1.5 segundos de inmunidad a colisiones de pared** para evitar que quede atrapado de forma infinita en esquinas o pasillos estrechos[4].
* **Estrategia Espacial en Dunas:** Los pasillos laterales cerrados de Dunas aumentan el riesgo de sufrir un *Wall Bang*, mientras que la pirámide central escalonada proporciona espacio abierto pero deja al jugador expuesto a proyectiles y ataques en área
